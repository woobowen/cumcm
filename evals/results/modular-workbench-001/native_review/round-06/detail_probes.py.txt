import copy

sys.dont_write_bytecode = True
ROOT = Path(__file__).resolve().parents[4]
OUT = Path(__file__).resolve().parent
SOURCE = ROOT / 'scripts/check_modular_workbench.py'
start = datetime.datetime.now(datetime.timezone.utc)
RUN = OUT / ('detail-' + start.strftime('%Y%m%dT%H%M%S%fZ'))
fixture = RUN / 'fixture'
fixture.mkdir(parents=True)
sha = lambda b: hashlib.sha256(b).hexdigest()

spec = importlib.util.spec_from_file_location('qualification_details_round05', SOURCE)
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
results = {}

def write(name, data):
    relative = m.BASE + '/' + name
    path = fixture / relative
    path.parent.mkdir(parents=True, exist_ok=True)
    data = data if isinstance(data, bytes) else json.dumps(data).encode()
    path.write_bytes(data)
    return {'path': relative, 'sha256': sha(data)}

def observe(name, action):
    try:
        results[name] = {'returned': action()}
    except Exception as exc:
        results[name] = {'exception': type(exc).__name__, 'message': str(exc)}

command = {'argv': ['ORIGINAL_NOT_EXECUTED'], 'exit_code': 42,
           'started_at': '2026-09-09T00:00:00Z', 'ended_at': '2026-09-09T00:00:01Z',
           'elapsed_seconds': 1, 'result': {'status': 'FAILED'}}
empty = write('original-empty.json', {})
code = write('original-not-executed.py', b'raise RuntimeError("ORIGINAL NOT EXECUTED")\n')
detail = {'schema_version': 'workbench-independent-recalculation/v1', 'implementation_hash': 'ORIGINAL',
          'commands': [command], 'producer_imported': False, 'independent_code': code,
          'results': [{'kind': kind, 'max_abs_residual': -1, 'numeric_values_checked': 3,
                       'record_packet': empty} for kind in ['prediction', 'optimization', 'mixed']]}
item = {'implementation_hash': 'ORIGINAL', 'detail': write('original-recalculation-detail.json', detail)}
observe('failed_recalculation_command_negative_abs_residual_empty_record_packets', lambda: m.validate_detail(fixture, 'independent_recalculation', item))

observe('empty_packet_after_valid_command_code_residual', lambda: m.validate_detail(fixture, 'independent_recalculation', empty_item))

    parsed = m.strict_json('{"elapsed_seconds":1e309}')
