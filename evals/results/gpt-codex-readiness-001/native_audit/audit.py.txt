"""Read-only, standard-library package/record/finite-arithmetic audit.

Does not import or execute supplied code. All writes stay beside this file.
Canonical JSON is independently implemented, using the package builder's
documented sorted UTF-8 JSON convention (no newline in digest).
"""
from pathlib import Path
from fractions import Fraction as F
import ast
import hashlib
import json
import math
import re

B = Path('<REPO_ROOT>')
O = Path(__file__).parent
R = B / 'evals/results/gpt-codex-readiness-001'
S = B / '.cache/gpt-codex-readiness-001/input_snapshot'
reads, checks, details = [], {}, {}


def h(data):
    return hashlib.sha256(data).hexdigest()


def canon(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=False,
                      separators=(',', ':'), allow_nan=False).encode()


def read(path, purpose='programmatic verification'):
    data = path.read_bytes()
    reads.append({'path':str(path), 'sha256':h(data), 'bytes':len(data), 'purpose':purpose})
    return data


def js(path):
    return json.loads(read(path))


for tag in ('start','next'):
    path=B/f'.cache/gpt-codex-readiness-001/{tag}-audit.zip'
    read(path, 'archive hash and member integrity')
    with zipfile.ZipFile(path) as z:
        names=z.namelist()
        assert len(names)==len(set(names))
        assert z.testzip() is None
        manifest=json.loads(z.read('PACKAGE_MANIFEST.json'))
        assert set(names)-{'PACKAGE_MANIFEST.json'}==set(manifest['files'])
        for name,digest in manifest['files'].items():
            data=z.read(name)
            assert h(data)==digest
            assert data==read(O/tag/name, 'archive content; hashed/parsed or separately text-reviewed')
        assert h(canon(manifest['files']))==manifest['payload_set_sha256']
        checks[f'{tag}_outer_integrity']=True
        details[f'{tag}_identity']={'archive_sha256':h(path.read_bytes()),
            'kind':manifest['kind'],'members':len(names),
            'payload_set_sha256':manifest['payload_set_sha256']}

for tag,root in [('new',O/'next/review'),('old',S/'original_example_review')]:
    m=js(root/'manifest.json')
    for name,digest in m['files'].items():
        assert h(read(root/name))==digest
    assert h(canon({k:v for k,v in m.items() if k!='package_hash'}))==m['package_hash']
    available={}
    for v in m['views']:
        data=read(root/v['path'])
        assert h(data)==v['view_sha256']
        available[v['source_path']]={'data':data,'view_hash':h(data),
                                    'source_hash':v['source_sha256']}
    checks[f'{tag}_review_integrity']=True
    details[f'{tag}_review']={'case_id':m['case_id'],'package_hash':m['package_hash'],
       'module':m['module'],'revision':m['revision'],
       'available_source_views':len(available),'referenced_sources':len(m['original_hashes']),
       'source_bytes_not_available_here': sorted(set(m['original_hashes'])-set(available))}
    raw=json.loads(available['data/raw/input.json']['data'])
    plan=json.loads(available['experiments/experiment_plan.json']['data'])['content']
    rows={r['observation_id']:r for r in raw['observations']}
    assert len(rows)==len(raw['observations'])==13
    ends={}
    # Independent exact OLS using normal-equation sums, different from endpoint
    # checker and producer centered-float implementation.
    for s in plan['temporal_design']['samples']:
        pts=[rows[k] for k in s['feature_observation_ids']]
        n=len(pts);x=[F(p['observed_at']) for p in pts];y=[F(str(p['value'])) for p in pts]
        slope=(n*sum(a*b for a,b in zip(x,y))-sum(x)*sum(y))/(n*sum(a*a for a in x)-sum(x)**2)
        intercept=(sum(y)-slope*sum(x))/n
        assert all(a*slope+intercept==b for a,b in zip(x,y))
        assert slope==F(-1,10)
        for key in ('feature_observation_ids','preprocess_fit_observation_ids','model_fit_observation_ids'):
            assert all(rows[k]['observed_at']<=s['origin'] and rows[k]['available_at']<=s['origin'] for k in s[key])
        end=(F(str(raw['threshold']))-intercept)/slope
        ends[s['sample_id']]=end
        if s['split']=='FORECAST':
            assert s['target_observation_id'] is None
            remaining=end-s['origin']
        else:
            assert end==rows[s['target_observation_id']]['observed_at']
    demand=math.ceil(remaining*F(str(raw['flow_litres_per_minute'])))
    def optimum(d):
        # Positive-cost feasible solution gives finite exhaustive rectangle.
        capacities=[c['litres'] for c in raw['containers']]
        prices=[c['cost'] for c in raw['containers']]
        bound=math.ceil(F(d,capacities[0]))*prices[0]
        feasible=[(i*prices[0]+j*prices[1],i,j,i*capacities[0]+j*capacities[1])
                  for i in range(bound//prices[0]+1)
                  for j in range(bound//prices[1]+1)
                  if i*capacities[0]+j*capacities[1]>=d]
        return min(feasible)
    expected=6 if tag=='new' else 8
    assert remaining==expected and demand==expected
    best=optimum(demand); perturb=optimum(demand+1)
    outputs={}
    for source,v in available.items():
        if source.startswith('runs/') and source.endswith('/output.json'):
            out=json.loads(v['data']);a=out['allocation'];counts=a['counts']
            assert a['demand_litres']==demand
            assert all(type(k) is int and k>=0 for k in counts)
            assert a['quantity_litres']==sum(k*c['litres'] for k,c in zip(counts,raw['containers']))
            assert a['cost']==sum(k*c['cost'] for k,c in zip(counts,raw['containers']))
            assert a['balance_litres']==[a['quantity_litres']-t for t in range(demand+1)]
            assert a['quantity_litres']>=demand
            if out['candidate_id']=='CAND':assert (a['cost'],*counts,a['quantity_litres'])==best
            else:assert counts==[math.ceil(F(demand,3)),0]
            samples=out['metric_samples']['metric_b']
            score=sum(abs(s['predicted_end_time']-s['observed_end_time'])/(s['observed_end_time']-s['origin'])*100 for s in samples)/len(samples)
            assert score==out['validation_metrics']['metric_b']
            assert abs(score)<1e-9
            prediction=out['scientific_evidence']['REQ-B']['prediction_evidence']
            assert abs(prediction['predictions'][0]['value']-expected)<1e-9
            assert prediction['empirical_accuracy_verified'] is False
            sens=out['robustness_evidence']['perturbations'][0]['result']
            expected_sens=math.ceil(F(demand+1,3))*4 if out['candidate_id']=='BASE' else perturb[0]
            assert sens==expected_sens
            outputs[out['candidate_id']]={'allocation':a,'metric_b':score,'perturbed_cost':sens,'sha256':v['view_hash']}
    handoff=json.loads(available['handoff/modeling_to_paper.json']['data'])
    for cid,c in handoff['claim_evidence'].items():
        rid=c['requirement_id'];run=c['selected_run_ids'][0]
        out=json.loads(available[f'runs/{run}/output.json']['data'])
        if tag=='new':
            assert c['statement']==out['requirement_claims'][rid]['claim_text']
            assert re.search(r'[\u4e00-\u9fff]',c['statement'])
            assert c['selected_run_ids']==['RUN-BASE-20260910']
        else:assert c['statement']==f'Bounded result for {rid}.'
    assert handoff['sources'][0]['hash']==h(available['data/raw/input.json']['data'])
    checks[f'{tag}_finite_arithmetic_and_claim_binding']=True
    details[f'{tag}_arithmetic']={'conditional_remaining_minutes':expected,'demand_litres':demand,
        'optimum_cost_counts_quantity':best,'perturbed_optimum':perturb,
        'outputs':outputs,'exact_historical_mre':0,'historical_entities':1,'historical_origins':2,
        'future_truth_observed':False}
    if tag=='new':
        assert plan['selection_rule']=='ARGMIN_THEN_ID'
        assert min((v['allocation']['cost'],k) for k,v in outputs.items())==(8,'BASE')
        assert set(handoff['symbols'])>={'T','t','t0','tau','b','q','s','cost','n3','n5','d'}
        assert available['models/runtime_model.py']['data']!=available['models/independent_check.py']['data']
        # Static parse only, no supplied code execution or importing.
        for source in ('models/runtime_model.py','models/independent_check.py'):
            ast.parse(available[source]['data'])
        checker_ast=ast.parse(available['models/independent_check.py']['data'])
        imports=[n.module for n in ast.walk(checker_ast) if isinstance(n,ast.ImportFrom)]
        imports += [a.name for n in ast.walk(checker_ast) if isinstance(n,ast.Import) for a in n.names]
        assert not any('runtime_model' in i for i in imports)
        details['checker_imports']=imports
        captures=[]
        for src,v in available.items():
            if src.endswith('capture.json'):
                d=json.loads(v['data'])
                assert d['exit_code']==0
                for p,digest in d.get('bound_files',{}).items():
                    if p in available:assert available[p]['source_hash']==digest
                if 'output' in d:assert available[d['output']['path']]['source_hash']==d['output']['sha256']
                for cf in d.get('code_files',[]):
                    if cf['path'] in available:assert available[cf['path']]['source_hash']==cf['sha256']
                captures.append({'source':src,'mode':d['capture_mode'],'run':d['run_id'],
                                 'started_at':d['started_at'],'ended_at':d['ended_at']})
        ledger=json.loads(available['evidence/scientific_final_ledger.json']['data'])
        assert ledger['count']==1 and ledger['test_access_count']==0 and ledger['status']=='SUCCESS'
        assert len(ledger['checks'])==1
        for run,check in ledger['checks'].items():
            assert check['exit_code']==0
            for p,digest in check['files'].items():
                if p in available:assert available[p]['source_hash']==digest
        details['captures']=captures
        details['final_ledger']=ledger
