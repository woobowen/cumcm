# Verify static core call signatures without importing the core module.
signatures={}
for name,node in definitions[CORE].items():
    args=node.args
    pos=args.posonlyargs+args.args
    defaults=[inspect.Parameter.empty]*(len(pos)-len(args.defaults))+[None]*len(args.defaults)
    pars=[]
    for i,(arg,default) in enumerate(zip(pos,defaults,strict=True)):
        kind=inspect.Parameter.POSITIONAL_ONLY if i<len(args.posonlyargs) else inspect.Parameter.POSITIONAL_OR_KEYWORD
        pars.append(inspect.Parameter(arg.arg,kind,default=default))
    if args.vararg: pars.append(inspect.Parameter(args.vararg.arg,inspect.Parameter.VAR_POSITIONAL))
    for arg,default in zip(args.kwonlyargs,args.kw_defaults,strict=True):
        pars.append(inspect.Parameter(arg.arg,inspect.Parameter.KEYWORD_ONLY,default=inspect.Parameter.empty if default is None else None))
    if args.kwarg: pars.append(inspect.Parameter(args.kwarg.arg,inspect.Parameter.VAR_KEYWORD))
    signatures[name]=inspect.Signature(pars)
calls=[]
for node in ast.walk(trees[PREP]):
    if isinstance(node,ast.Call) and isinstance(node.func,ast.Attribute) and isinstance(node.func.value,ast.Name) and node.func.value.id=="core":
        name=node.func.attr
        record={"function":name,"line":node.lineno}
        if name not in signatures: record["result"]="NO_LOCAL_FUNCTION_DEFINITION"
        elif any(isinstance(a,ast.Starred) for a in node.args) or any(k.arg is None for k in node.keywords): record["result"]="DYNAMIC_ARGUMENTS_NOT_CHECKED"
        else:
            try:
                signatures[name].bind(*[None]*len(node.args),**{k.arg:None for k in node.keywords})
                record["result"]="ARGUMENT_SHAPE_MATCH"
            except TypeError as exc: record["result"]="MISMATCH:"+str(exc)
        calls.append(record)
