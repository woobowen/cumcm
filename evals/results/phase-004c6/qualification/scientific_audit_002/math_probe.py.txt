def exact_forecast(data,state,size):
    x=[sum(map(Fraction,values))/state for values in zip(*data[:state],strict=True)]
    y=list(map(Fraction,data[state][:size]))
    mx=sum(x[:size])/size
    my=sum(y)/size
    b=sum((u-mx)*(v-my) for u,v in zip(x[:size],y,strict=True))/sum((u-mx)**2 for u in x[:size])
    a=my-b*mx
    return a+b*x[-1]
