def functions(path, names, namespace):
    tree = ast.parse((SNAP / path).read_text())
    code = ast.Module(body=[n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names], type_ignores=[])
    assert {n.name for n in code.body} == set(names)
    exec(compile(code, path, "exec", flags=__import__('__future__').annotations.compiler_flag), namespace)
    return namespace

def write(path, value, **kwargs):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2) + "\n")

def error_of(call):
    try:
        call()
    except ValueError as exc:
        return str(exc)
    return None

# Numeric helper cross-check beyond the four existing tests: unmatched knots,
# durations and current weights; independently assembled scalar checker curve.
del sys.modules["openpyxl"]
import numpy as np
from scipy.interpolate import PPoly, PchipInterpolator
producer_path = "evals/results/phase-004c6/CUMCM-2016-C-POSTVALIDATION-DEVELOPMENT-006/code/produce.py"
checker_path = "evals/results/phase-004c6/CUMCM-2016-C-POSTVALIDATION-DEVELOPMENT-006/code/check.py"
producer = functions(producer_path, {"surface", "crossing"}, {"np": np, "PPoly": PPoly})
checker = functions(checker_path, {"coefficients", "evaluate", "inverse"}, {"math": math, "bisect": bisect})
numeric = []
for cubic in (False, True):
    sets = {50: [(0,12),(0.7,9),(1.9,11),(3,8)], 60: [(0,12),(2.2,9.3),(3.8,11.3),(6,8)]}
    polys, scalar = {}, {}
    for current, points in sets.items():
        x = np.array([p[0] for p in points]); y = np.array([p[1] for p in points])
        polys[current] = PchipInterpolator(x, y) if cubic else PPoly(np.vstack([np.zeros(3),np.zeros(3),np.diff(y)/np.diff(x),y[:-1]]),x)
        c = checker["coefficients"](points, cubic)
        def f(t,x=x,c=c): return checker["evaluate"](x,c,t)
        def d(t,x=x,c=c):
            j=min(max(0,bisect.bisect_right(x,t)-1),len(c)-1);a,b,c1,_=c[j];z=t-x[j]
            return (3*a*z+2*b)*z+c1
        f.knots=x.tolist(); f.derivative=d;scalar[current]=f
    candidate = "CUBIC_LOG_AFFINE" if cubic else "BASELINE"
    forward, inverse, duration, metadata = producer["surface"](polys,{50:3,60:6},55,candidate)
    w = metadata["weight"]
    def mixed(t): return (1-w)*scalar[50](t*3/duration)+w*scalar[60](t*6/duration)
    mixed.knots=sorted(set([t*duration/3 for t in scalar[50].knots]+[t*duration/6 for t in scalar[60].knots]))
    mixed.derivative=lambda t:(1-w)*scalar[50].derivative(t*3/duration)*3/duration+w*scalar[60].derivative(t*6/duration)*6/duration
    residuals=[]
    for voltage in (8.1,9,9.8,10,10.5,11.0):
        a=inverse(voltage); b=checker["inverse"](mixed,duration,voltage)
        assert abs(a-b)<1e-9,(candidate,voltage,a,b)
        assert abs(forward(a/duration)-voltage)<1e-9
        residuals.append(abs(a-b))
    numeric.append({"candidate":candidate,"voltages":6,"max_inverse_difference_min":max(residuals)})
RESULTS["independent_numeric"] = numeric
write(HERE / "probe_results.json", RESULTS)
print(json.dumps(RESULTS, ensure_ascii=False, indent=2))
