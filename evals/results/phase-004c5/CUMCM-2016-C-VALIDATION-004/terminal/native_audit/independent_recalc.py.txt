def evalpoly(rec, t):
    x=np.asarray(rec['time_knots_min']); c=np.asarray(rec['coefficients_descending'])
    j=np.clip(np.searchsorted(x,t,side='right')-1,0,len(x)-2)
    d=t-x[j]
    return ((c[0,j]*d+c[1,j])*d+c[2,j])*d+c[3,j]

def inverse_segments(rec, u):
    x=np.asarray(rec['time_knots_min']); c=np.asarray(rec['coefficients_descending']); hits=[]
    for j in range(len(x)-1):
        if evalpoly(rec,x[j]) >= u-1e-10 and evalpoly(rec,x[j+1]) <= u+1e-10:
            l,h=float(x[j]),float(x[j+1])
            for _ in range(65):
                m=(l+h)/2
                if evalpoly(rec,m)>u:l=m
                else:h=m
            hits.append((l+h)/2)
    if not hits:raise AssertionError('no segment')
    return max(hits)

def aff_existing(x,y,affine):
    xp=x[:len(y)]
    if affine:
        b=float(np.sum((xp-xp.mean())*(y-y.mean()))/np.sum((xp-xp.mean())**2));a=float(y.mean()-b*xp.mean())
    else:a=0.;b=float(np.sum(xp*y)/np.sum(xp*xp))
    return a,b,a+b*x
