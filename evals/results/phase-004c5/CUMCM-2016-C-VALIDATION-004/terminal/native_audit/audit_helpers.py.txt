from pathlib import Path
import hashlib,json,datetime
ROOT=Path('<REPO_ROOT>')
OUT=ROOT/'.cache/pr12-rc8/fresh/2016/native_terminal_audit'
def read(p,purpose='content'):
 p=Path(p); p=p if p.is_absolute() else ROOT/p
 b=p.read_bytes()
 with (OUT/'read_events.jsonl').open('a') as f: f.write(json.dumps({'path':str(p.relative_to(ROOT)),'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b),'purpose':purpose,'read_at':datetime.datetime.now(datetime.timezone.utc).isoformat()})+'\n')
 return b
def show(p):
 print('FILE',p); print(read(p).decode())
